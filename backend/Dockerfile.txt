# ─── AirGuard Backend – Dockerfile Multi-Stage ──────────────────
# AirGuard – Portal web de calidad del aire y salud ambiental El Salvador
# Stage 1: Builder — instala dependencias
FROM node:20-alpine AS builder

WORKDIR / app

# Copiar manifests primero para aprovechar cache de capas
COPY package*.json ./
RUN npm ci --only=production && npm cache clean --force

# Copiar código 
COPY src/ ./src/

# ─── Stage 2: Production ────────────────────────────────────────
FROM node:20-alpine AS production

WORKDIR /app

# Metadatos
LABEL maintainer="AirGuard Team"
LABEL version="0.9.5"
LABEL description="AirGuard API – Monitoreo de Calidad del Aire, San Salvador"

ENV NODE_ENV=production

# Solo lo necesario del stage anterior
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/src ./src
COPY package*.json ./

# Usuario no-root por seguridad
RUN addgroup -S airguard && adduser -S airguarduser -G airguard
USER airguarduser

EXPOSE 3001

# Health check para Render/Docker
HEALTHCHECK --interval=30s --timeout=10s --start-period=10s --retries=3 \
  CMD wget -qO- http://localhost:3001/health || exit 1

CMD ["node", "src/index.js"]
